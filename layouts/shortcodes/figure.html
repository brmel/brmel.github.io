{{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- $img := .Page.Resources.GetMatch $src -}} {{- $processable := and $img (in (slice "jpg" "jpeg" "png" "webp" "tif" "bmp" "gif") $img.MediaType.SubType) -}} {{- /* An image is never laid out wider than it really is. srcset `w` descriptors make the browser density-correct the intrinsic size to whatever `sizes` claims, so a hardcoded sizes made every figure lay out at that width — a 223px screenshot was being stretched to 960px. The display width is therefore capped at the asset's own width, and `sizes` states that same number. */ -}} {{- $small := "" }}{{ $large := "" }}{{ $displayW := 0 }} {{- if $processable }} {{- $w := $img.Width }} {{- $large = cond (gt $w 1600) ($img.Resize "1600x webp q82") ($img.Resize (printf "%dx webp q82" $w)) }} {{- $small = cond (gt $w 800) ($img.Resize "800x webp q82") $large }} {{- $col := site.Params.proseColumnPx | default 900 -}} {{- $displayW = cond (gt $large.Width $col) $col $large.Width }} {{- end }} {{- with .Get "link" }}{{ end -}} {{- if $processable }} {{- else }} {{- with $img }}{{ $src = .RelPermalink }}{{ end }} {{- end }} {{- with .Get "link" }}{{ end }} {{- if or (or (.Get "title") $caption) (.Get "attr") }} {{- end }}
