{{- /* figure — local fork of PaperMod's shortcode. Upstream resolves a page resource and emits a bare with no dimensions and no responsive sources. Article screenshots are the heaviest images on the site, so this version puts them through the same pipeline the project galleries use: * page-bundle images -> WebP at 800w and 1600w, with srcset * explicit width/height, so they reserve their space and cannot shift layout * remote or non-processable sources fall back to a plain , unchanged API is unchanged: src, alt, caption, title, attr, attrlink, link, target, rel, align, class, width, height. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- $img := .Page.Resources.GetMatch $src -}} {{- $processable := and $img (in (slice "jpg" "jpeg" "png" "webp" "tif" "bmp" "gif") $img.MediaType.SubType) -}} {{- $small := "" }}{{ $large := "" }} {{- if $processable }} {{- $w := $img.Width }} {{- $large = cond (gt $w 1600) ($img.Resize "1600x webp q82") ($img.Resize (printf "%dx webp q82" $w)) }} {{- $small = cond (gt $w 800) ($img.Resize "800x webp q82") $large }} {{- end }} {{- with .Get "link" }}{{ end -}} {{- if $processable }} {{- else }} {{- with $img }}{{ $src = .RelPermalink }}{{ end }} {{- end }} {{- with .Get "link" }}{{ end }} {{- if or (or (.Get "title") $caption) (.Get "attr") }} {{- end }}
