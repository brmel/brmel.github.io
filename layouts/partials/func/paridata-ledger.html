{{- $profile := site.Data.paridata.profile -}} {{- $competitions := site.Data.paridata.competitions -}} {{- $flags := slice "england" "spain" "italy" "france" "germany" "netherlands" "portugal" "europe" "mix" -}} {{- $tickets := slice -}} {{- range $name, $file := site.Data.paridata -}} {{- with $file.tickets -}}{{- $tickets = $tickets | append . -}}{{- end -}} {{- end -}} {{- $rows := slice -}} {{- $balance := 0.0 -}} {{- $staked := 0.0 -}} {{- $settledCount := 0 -}} {{- range $t := sort $tickets "postedAt" -}} {{- $stake := $t.stake | default $profile.staking.defaultStake -}} {{- $regions := slice -}} {{- $voidOdds := 1.0 -}} {{- range $leg := $t.legs -}} {{- with index $competitions ($leg.competition | default "") -}}{{- $regions = $regions | append . -}}{{- end -}} {{- if eq $leg.settlement "void" -}}{{- $voidOdds = mul $voidOdds $leg.odds -}}{{- end -}} {{- end -}} {{- $regions = uniq $regions -}} {{- $region := cond (eq (len $regions) 1) (index $regions 0) "mix" -}} {{- if not (in $flags $region) -}}{{- $region = "mix" -}}{{- end -}} {{- $profit := 0.0 -}} {{- $settled := ne $t.settlement "pending" -}} {{- if eq $t.settlement "won" -}} {{- $profit = mul $stake (sub (div $t.odds $voidOdds) 1.0) -}} {{- else if eq $t.settlement "lost" -}} {{- $profit = sub 0.0 $stake -}} {{- end -}} {{- if in (slice "won" "lost") $t.settlement -}}{{- $staked = add $staked $stake -}}{{- end -}} {{- if $settled -}} {{- $balance = add $balance $profit -}} {{- $settledCount = add $settledCount 1 -}} {{- end -}} {{- $rows = $rows | append (dict "ticket" $t "postedAt" $t.postedAt "region" $region "kind" (cond (gt (len $t.legs) 1) "coupon" "single") "legs" (len $t.legs) "stake" $stake "hasVoid" (ne $voidOdds 1.0) "profit" $profit "settled" $settled "balance" $balance) -}} {{- end -}} {{- return (dict "profile" $profile "rows" $rows "balance" $balance "staked" $staked "settledCount" $settledCount "roi" (cond (gt $staked 0.0) (mul (div $balance $staked) 100.0) 0.0)) -}}
